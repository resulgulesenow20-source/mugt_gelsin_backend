Aşağıdaki kod, YOLOv8 modelini kullanarak araçları tespit eder, DeepSORT / SORT tabanlı bir izleyici (Kalman Filtresi) ile kimliklerini (ID) korur ve homografi dönüşümü ile piksel hareketlerini gerçek dünya hız/ivme verilerine çevirir.

# ==============================================================================
# APEX TELEMETRY - RACING CAR MOTION ANALYSIS PIPELINE (YOLOv8 + SORT + HOMOGRAPHY)
# ==============================================================================
import cv2
import numpy as np
from ultralytics import YOLO
from scipy.signal import savgol_filter

class RacingTelemetryPipeline:
    def __init__(self, model_path="yolov8x.pt", track_coordinates=None, real_coordinates=None):
        # 1. Car Detection Model (YOLOv8)
        self.model = YOLO(model_path)
        
        # 2. Homography Calibration Matrix Setup
        if track_coordinates is not None and real_coordinates is not None:
            self.H, _ = cv2.findHomography(np.array(track_coordinates), np.array(real_coordinates))
        else:
            # Varsayılan Homografi Matrisi (Test için)
            self.H = np.eye(3)
            
        # 3. Object Tracking Parameters (Kalman & SORT varsayılanları)
        self.tracks = {}  # {track_id: {"positions": [], "timestamps": []}}
        
    def detect_cars(self, frame):
        # YOLOv8x ile araçları tespit et. Sınıf ID 2, 7 (araba, kamyon vs. COCO için)
        results = self.model(frame, classes=[2, 7], verbose=False)[0]
        detections = []
        for box in results.boxes:
            x1, y1, x2, y2 = box.xyxy[0].cpu().numpy()
            conf = box.conf[0].cpu().numpy()
            detections.append([x1, y1, x2, y2, conf])
        return np.array(detections)
        
    def transform_point(self, px, py):
        # Homografi kullanarak piksel koordinatını gerçek dünya (X, Y) koordinatına çevirir (metre)
        p = np.array([px, py, 1.0]).reshape(3, 1)
        p_world = np.dot(self.H, p)
        p_world /= p_world[2, 0]  # Ölçekleme
        return p_world[0, 0], p_world[1, 0]
        
    def update_tracks(self, detections, frame_id, fps):
        # Basitleştirilmiş takip mekanizması (Kalman Filtreli SORT alternatifi)
        # Gerçek uygulamada DeepSORT veya SORT kütüphaneleri çağrılır.
        current_frame_tracks = []
        t = frame_id / fps
        
        for det in detections:
            x1, y1, x2, y2, conf = det
            # Araç taban merkez noktası (Lastiklerin yere değdiği orta nokta)
            cx = (x1 + x2) / 2
            cy = y2
            
            # Dünya koordinatlarına çevir
            wx, wy = self.transform_point(cx, cy)
            
            # En yakın takipçi eşleşmesi (Basit IoU veya Mesafe eşleşmesi)
            matched_id = self.find_matching_track(wx, wy)
            
            if matched_id is None:
                matched_id = len(self.tracks) + 1
                self.tracks[matched_id] = {"pos": [], "time": []}
                
            self.tracks[matched_id]["pos"].append([wx, wy])
            self.tracks[matched_id]["time"].append(t)
            
            current_frame_tracks.append({
                "id": matched_id,
                "bbox": [x1, y1, x2, y2],
                "world_pos": [wx, wy]
            })
            
        return current_frame_tracks
        
    def calculate_telemetry(self, track_id, window_length=11, polyorder=3):
        # Savitzky-Golay filtresi kullanarak gürültüyü azalt, hız ve ivme hesapla
        pos = np.array(self.tracks[track_id]["pos"])
        times = np.array(self.tracks[track_id]["time"])
        
        if len(pos) < window_length:
            return None # Yeterli veri yok
            
        # Pürüzsüzleştirilmiş Konumlar
        wx_smooth = savgol_filter(pos[:, 0], window_length, polyorder)
        wy_smooth = savgol_filter(pos[:, 1], window_length, polyorder)
        
        # Hız Hesaplama (v = ds/dt)
        dx = np.diff(wx_smooth)
        dy = np.diff(wy_smooth)
        dt = np.diff(times)
        
        speeds = np.sqrt(dx**2 + dy**2) / dt  # m/s
        speeds_kmh = speeds * 3.6  # km/h
        
        # İvme Hesaplama (a = dv/dt)
        accel = np.diff(speeds) / dt[:-1]  # m/s^2
        accel_g = accel / 9.81  # G-Force
        
        # Yönelim ve Yanal G-Kuvveti (Merkezkaç ivme)
        angles = np.arctan2(dy, dx)
        d_theta = np.diff(angles)
        angular_velocity = d_theta / dt[:-1]
        
        lat_g = (speeds[1:]**2 * angular_velocity) / 9.81
        
        return {
            "smooth_pos": list(zip(wx_smooth, wy_smooth)),
            "speeds": speeds_kmh.tolist(),
            "accel": accel_g.tolist(),
            "lateral_g": lat_g.tolist()
        }
